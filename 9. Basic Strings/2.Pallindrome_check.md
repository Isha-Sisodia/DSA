### Method - 1
```python
class Solution:
    def isPalindrome(self, s):
        i = 0
        j = len(s) - 1
        while i < j:
            if s[i] != s[j]:
                return False
            i += 1
            j -= 1
        return True
```
### Method - 2 Using Helper function
```python
class Solution:
    def isPalindrome(self, s):
        def helper(i,j):
            if i >= j:
                return True
            if s[i] != s[j]:
                return False
            return helper(i+1, j-1)
        return helper(0,len(s)-1)
```
