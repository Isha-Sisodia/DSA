# Second Largest Element
```python
class Solution:
    def getSecondLargest(self, arr):
        # code here
        max_1 = -1         #float("-inf")
        max_2 = -1         #float("-inf")
        for num in arr:
            if num > max_1:
                max_2 = max_1
                max_1 = num
            elif num == max_1:   #Important for example - arr= [10,5,10] 
                continue
            else:
                max_2 = max(max_2, num)
        return max_2
```
### or
```python
class Solution:
    def getSecondLargest(self, arr):
        max_1 = float("-inf")   # max_1 = -1
        max_2 = float("-inf")   # max_2 = -1

        for num in arr:
            if num > max_1:
                max_2 = max_1
                max_1 = num

            elif num != max_1:
                max_2 = max(max_2, num)

        return max_2
```

# Second Largest and Second Smallest Element
**This Approach can easily be used for finding third Largest/Smallest element and so on.** by adding another for loops.
```python
class Solution:
    def getSecondLargest(self, arr):
        # code here
        max_1 = -1       # float("-inf")  
        max_2 = -1       # float("-inf")  
        min_1 = float("inf")
        min_2 = float("inf")
        for i in range(len(arr)):
            min_1 = min(min_1, arr[i])
            max_1 = max(max_1, arr[i])
        for i in range(len(arr)):
            if arr[i] > max_2 and arr[i] != max_1:
                max_2 = arr[i]
            if arr[i] < min_2 and arr[i] != min_1:
                min_1 = arr[i]
        return max_2
```
