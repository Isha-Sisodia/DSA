        i = 0

### Method - 2 - Using Hashing
```python
class Solution:
    def findUnion(self, a, b):
        c = []
        freq = {}
        for num in a:
            if num in freq:
                freq[num] += 1
            else:
                freq[num] = 1
        for num in b:
            if num in freq:
                freq[num] += 1
            else:
                freq[num] = 1
        for num in freq:
            c.append(num)
            
        c.sort()
        
        return c
```
### Method 3 - Using set
```python
class Solution:
    def findUnion(self, a, b):
        c = set(a) | set(b)
        return sorted(c)    #sorted(c) automatically returns a list in Python
```
