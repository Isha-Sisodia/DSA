```python
class Solution(object):
    def rearrangeArray(self, nums):
        positive = []
        negative = []
        for i in range(len(nums)):
            if nums[i] > 0:
                positive.append(nums[i])
            else:
                negative.append(nums[i])
        answer = []
        for i in range(len(positive)):
            answer.append(positive[i])
            answer.append(negative[i])
        return answer
```
