```python
class Solution:
    def majorityElement(self, arr):
        n = len(arr)//2
        freq = {}
        for num in arr:
            if num not in freq:
                freq[num] = 1
            else:
                freq[num] += 1
        for num in freq:
            if freq[num] > n:
                return num
        return -1
```
