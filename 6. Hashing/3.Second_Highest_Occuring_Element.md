class Solution:
    def secondMostFrequentElement(self, arr):
        freq = {}

        # Count frequency
        for num in arr:
            freq[num] = freq.get(num, 0) + 1

        # Find highest frequency
        max_freq = max(freq.values())

        # Find second highest DISTINCT frequency
        second_freq = -1

        for f in freq.values():
            if f < max_freq:
                second_freq = max(second_freq, f)

        # No second frequency exists
        if second_freq == -1:
            return -1

        # Find smallest element having second highest frequency
        ans = float('inf')

        for num in freq:
            if freq[num] == second_freq:
                ans = min(ans, num)

        return ans
