### Method 1
```python
name = "Isha"
n = 3
def print_name(name,n):

    #Base Condition
    if n == 0:
        return
    print(name, end = " ")
    n-=1
    print_name(name,n)
    
#Function Calling    
print_name(name,n)
```
### Method 2
```python
n = int(input("Enter a Number: "))
name = input("Enter your Name: ")
# count = 0  #No Need
def print_name(name,count):
    if count == n:
        return
    print(name)
    print_name(name,count+1)
    
#Function Calling    
print_name(name,0) #here 0 is count
```
