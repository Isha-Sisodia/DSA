class Solution:
    def nthFibonacci(self, n: int) -> int:
        # code here
        a = 0
        b = 1
        count = 0
        ans = []
        sum = 0
        while count <= n:    #Important
            ans.append(a)
            sum = a + b
            a = b
            b = sum
            count += 1
        return ans[n]
