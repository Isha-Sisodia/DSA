class Solution:
    def commonElements(self,a,b):
        result = []
        freq = {}
        for num in a:
            if num in freq:
                freq[num] += 1
            else:
                freq[num] = 1
        for num in b:
            if num in freq and freq[num] > 0:
                result.append(num)
                freq[num] -= 1
            result.sort()
        return result
```

class Solution:
    def commonElements(self,a,b):
        a.sort()
        b.sort()
        n1 = len(a)
        n2 = len(b)
        i = 0
        j = 0
        result = []
        while i < n1 and j < n2:
            if a[i] == b[j]:
                result.append(a[i])
                i += 1
                j += 1
            elif a[i] < b[j]:
                i += 1
            else:
                j += 1
            
        return result
```
