https://www.geeksforgeeks.org/problems/reverse-a-string/1
```python
class Solution:
    def reverseString(self, s: str) -> str:
        s = list(s)       #Important
        i = 0
        j = len(s) - 1
        while i < j:
            s[i], s[j] = s[j], s[i]
            i += 1
            j -= 1
        # ans = str(s)   #not necessary
        return "".join(s)
```
