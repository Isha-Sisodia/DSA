```python
class Solution:
	def pushZerosToEnd(self, a):
	    temp = [0]*len(a)
	    idx = 0
	    for num in a:
	        if num != 0:
	            temp[idx] = num
	            idx += 1
	    i = 0
	    for num in temp:
	        a[i] = num
	        i += 1
	    return a
```
