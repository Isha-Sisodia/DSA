```python
class Solution:
    def missingNum(self, arr):
        # code here
        arr.sort()
        n = len(arr)
        for i in range(n):
            if arr[i] != i+1:
                return i + 1
        return n + 1
```

```python
class Solution:
    def missingNum(self, arr):
        n = len(arr)
        expected_sum = 0
        actual_sum = sum(arr)
        for i in range(1,n+2):
            expected_sum += i
        return expected_sum - actual_sum
```
