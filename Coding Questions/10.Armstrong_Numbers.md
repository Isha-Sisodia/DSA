```python
class Solution:
    def armstrongNumber (self, n):
        org = n
        cube = 0
        while n > 0:
            last_digit = n % 10
            cube += last_digit*last_digit*last_digit
            n//=10
            
        if cube == org:
            return True
        else:
            return False
```
