```python
class Solution:
    def intersection(self, arr1, arr2):
        i = 0
        j = 0
        n1 = len(arr1)
        n2 = len(arr2)
        result = []
        while i < n1 and j < n2:
            if arr1[i] < arr2[j]:
                i += 1
            elif arr1[i] == arr2[j]:
                # if arr1[i] not in result:    #it gives TLE
                if not result or result[-1] != arr1[i]:
                    result.append(arr1[i])
                i += 1
                j += 1
            else:
                j += 1
        return result
```
