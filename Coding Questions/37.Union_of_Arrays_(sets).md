```python
class Solution:    
    def findUnion(self, a, b):
        result = []
        for num in a:
            if num not in result:
                result.append(num)
        for num in b:
            if num not in result:
                result.append(num)
        return result
```

class Solution:    
    def findUnion(self, a, b):
        result = set(a) | set(b)
        result = list(result)
        return result
