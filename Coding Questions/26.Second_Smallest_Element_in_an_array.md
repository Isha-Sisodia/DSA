class Solution:
    def minAnd2ndMin(self, arr):
        min_1 = float("inf")         
        max_2 = float("inf")       
        for i in range(len(arr)):
            if arr[i] < min_1:
                min_2 = min_1
                min_1 = arr[i]
            if arr[i] < min_2 and arr[i] != min_1:
                min_2 = arr[i]
        if min_1 == float("inf") or min_2 == float("inf"):
            return [-1]
        return [min_1, min_2]
