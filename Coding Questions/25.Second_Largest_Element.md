class Solution:
    def getSecondLargest(self, arr):
        # code here
        max_1 = -1         
        max_2 = -1         
        for i in range(len(arr)):
            if arr[i] > max_1:
                max_2 = max_1
                max_1 = arr[i]
            if arr[i] > max_2 and arr[i] != max_1:
                max_2 = arr[i]
        return max_2
