class Solution:
    def primeRange(self, l, r):
        # code here
        c = []
        for n in range(l, r+1):
            is_prime = True
            if n < 2:
                is_prime = False
            for j in range(2,int(n**0.5)+1):
                if n % j == 0:
                    is_prime = False
                    break
            if is_prime:
                c.append(n)
        return c
