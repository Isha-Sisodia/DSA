```python
n = 5
space = 0

#Upper Half
for i in range(1,n+1):
    # space
    print(" " * space, end="")
    space += 1
    for j in range(i,n+1):
        print(j, end=" ")
        
    print()

#Lower Half    
for i in range(n-1,0,-1):
    #space
    print(" " * (i-1), end = "")
    for j in range(i, n+1):
        print(j, end = " ")
        
    print()
```
