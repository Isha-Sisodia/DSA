```python
n = 5
space  = (2*n)-2
stars = 2
for i in range(1,n+1):
    print("*" * (stars // 2) + space * " " + "*" * (stars // 2))
    space -= 2
    stars += 2
# stars = 12 #space = -2
space = space+2 #0      
stars = stars-2 #10
#or we can simply reinitialise them
#stars = 2*n
#space = 0
for i in range(n+1,2*n+1):
    space += 2
    stars -= 2
    print("*"*(stars//2)+space*" "+"*"*(stars//2))
```
