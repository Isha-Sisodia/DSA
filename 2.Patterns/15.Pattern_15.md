```python
# Method 1
n = 5

for i in range(n, 0, -1):
    for j in range(i):
        print(chr(65 + j), end="")
    print()
```

```python
# Method 2
n = 5 
for i in range(n,0,-1):
    som = 0
    for j in range(i):
        print(chr(ord("A")+som), end = "")
        som += 1
    print()
```
