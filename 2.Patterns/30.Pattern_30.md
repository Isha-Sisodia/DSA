```python
n = 5
space = 0
space_2 = n
#Upper Half
for i in range(n,0,-1):
    # space
    print(" " * space, end="")
    space += 1

    if i == n or i == 1:
        print("* " * i )
        
    else:
        print("*"+ " " * space_2 +"*")
        space_2 -=2
     
space = n-2  
space_2 = 1
#for lower half
for i in range(2,n+1):
    # space
    print(" " * space, end="")
    space -= 1

    if i == n:
        print("* " * i )
        
    else:
        print("*"+ " " * space_2 +"*")
        space_2 +=2
```
