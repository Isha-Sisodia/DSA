```python
n = 5
space = n + 2

for i in range(1, n + 1):

    # Left half
    for j in range(1, i + 1):
        print(j, end="")

    # Spaces
    print(" " * space, end="")
    space -= 2

    # Right half
    for j in range(i, 0, -1):
        print(j, end="")

    print()
```
