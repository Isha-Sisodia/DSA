```python
n = 5
space  = 0
stars = 2*n
for i in range(1,n+1):
    print("*" * (stars // 2) + space * " " + "*" * (stars // 2))
    space += 2
    stars -= 2

for i in range(n+1,2*n+1):
    space -= 2
    stars += 2
    print("*"*(stars//2)+space*" "+"*"*(stars//2))
    
```
