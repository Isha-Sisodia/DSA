### Method 1
```python
n = 4

for i in range(1, n + 1):

    # Leading spaces
    print("  " * (n - i), end="")

    # Descending numbers
    for j in range(i, 0, -1):
        print(j, end=" ")

    # Ascending numbers
    for j in range(2, i + 1):
        print(j, end=" ")

    print()
```
### Method 2
```python
n = 5
space = n-1
for i in range(1, n+1):
    print(" " * space, end = "")
    space -= 1
    for j in range(i,0,-1):
        print(j, end = "")
        
    for j in range(2,i+1):
        print(j, end = "")
    print()        
```
