### TLE Code - 
```python
class Solution:
    def gcd(self, n1, n2):
        # code here
        a = []
        b = []
        common = []
        for i in range(1, n1 + 1):
            if n1 % i == 0:
                a.append(i)
        for i in range(1, n2 + 1):
            if n2 % i == 0:
                b.append(i)
        for i in a:
            for j in b:
                if i==j:
                    common.append(i)
        ans  = max(common)
        return ans
```
