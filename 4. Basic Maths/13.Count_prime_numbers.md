```python
class Solution(object):
    def countPrimes(self, n):
        count = 0
        if n <= 1:
            return 0
        for i in range(2,n):  # which number are we checking?
            is_prime = True
            for j in range(2,int(i ** 0.5)+1):   # is i divisible by something?
                if i % j == 0:
                    is_prime = False
                    break
            if is_prime == True:
                count += 1
        return count
```
