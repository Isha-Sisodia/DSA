class Solution:
    def isPalindrome(self, n):
        # Method 1
        if n < 0:
            n = n - 2*n

        # or Method 2
        # if n < 0:
        #   n = - n

        # or Method 3
        # if n < 0:
            n = abs(n)
            
        original = n
		reverse = 0
		
		while n >0:
		    last_digit = n % 10
		    reverse = reverse * 10 + last_digit
		    n = n // 10
		    
		if reverse == original:
		    return True
		else:
		    return False
