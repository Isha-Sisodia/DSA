## Algorithm
1. Choose a pivot element.
2. Partition the array into two parts:
- Left: elements smaller than the pivot.
- Right: elements greater than the pivot.
3. Recursively apply Quick Sort to the left and right parts.
4. Combine the results (the array is sorted in place).
### Method - 1
```python
def quickSort(a, low, high):

    # Base case
    if low >= high:
        return

    pivot = a[low]

    i = low + 1
    j = high

    # Partition
    while i < j:

        # Move i to the right
        while i <= high and a[i] < pivot:
            i += 1

        # Move j to the left
        while j >= low and a[j] > pivot:
            j -= 1

        # Swap
        if i < j:
            a[i], a[j] = a[j], a[i]

    # Put pivot in its correct position
    a[low], a[j] = a[j], a[low]

    # Recursively sort left part
    quickSort(a, low, j - 1)

    # Recursively sort right part
    quickSort(a, j + 1, high)


a = [35, 50, 15, 25, 80, 20, 90, 45]

quickSort(a, 0, len(a) - 1)

print(a)
```
### Method 2 - Using two Functions
```python
def partition(a, low, high):

    pivot = a[low]

    i = low + 1
    j = high

    while i <= j:

        while i <= high and a[i] < pivot:
            i += 1

        while j >= low and a[j] > pivot:
            j -= 1

        if i < j:
            a[i], a[j] = a[j], a[i]

    # Put pivot in its correct position
    a[low], a[j] = a[j], a[low]

    return j


def quick_sort(a, low, high):

    if low >= high:
        return

    pivot_index = partition(a, low, high)

    quick_sort(a, low, pivot_index - 1)
    quick_sort(a, pivot_index + 1, high)


a = [6, 3, 8, 5, 2, 7, 4, 1]

quick_sort(a, 0, len(a) - 1)

print(a)
```
### Choose the middle element of the current subarray as the pivot.
But remember: Quick Sort does not require the pivot to always be the middle element. Pivot selection can be:
- first element
- last element
- middle element
- random element
In your current GFG code, we're choosing the middle element because **choosing the first element can cause TLE for already sorted arrays.**
### Method - 3 (TLE)
```python
class Solution:
    def quickSort(self, arr, low, high):
        # code here 
        if low>=high:
            return
        pivot_idx = self.partition(arr, low, high) 
        self.quickSort(arr, low, pivot_idx)
        self.quickSort(arr, pivot_idx+1,high)
        

    def partition(self, arr, low, high):
        # code here
        # pivot  = arr[low]
        pivot = arr[(low + high) // 2]
        i = low
        j = high
        while i <= j:
            while arr[i] < pivot:
                i += 1
            while arr[j] > pivot:
                j -= 1
            if i <= j :
                arr[i], arr[j] = arr[j], arr[i]
                i += 1
                j -= 1
        # arr[low], arr[j] = arr[j] , arr[low]
        return j
```
